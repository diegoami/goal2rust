"""Additional immutable post-writer provenance audit; run before validate.py."""
import hashlib,json,subprocess,sys
from pathlib import Path
sys.path.insert(0,'out/T05')
import evidence as e
out=e.BASE/'out/T05';e.verify();before=e.hashes()
assert not e.git('status','--porcelain',root=e.ORACLE)
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
harness={}
for p in sorted((e.ORACLE/'experiments/E036-match-port').glob('*.py'))+[e.ORACLE/'tests/test_port_seal.py']:
    rel=str(p.relative_to(e.ORACLE))
    blob=subprocess.check_output(['git','show',e.PIN+':'+rel],cwd=e.ORACLE)
    assert hashlib.sha256(blob).hexdigest()==sha(p)
    harness[rel]=sha(p)
lines=(out/'family-F40-rust-v1.jsonl').read_bytes().splitlines(keepends=True)
for n in [10,20,30]:
    p=out/f'family-progress-{n}-v1.jsonl';r=json.loads((out/f'family-progress-{n}-v1.json').read_text())
    assert p.read_bytes()==b''.join(lines[:n]) and sha(p)==r['sha256']
    assert [hashlib.sha256(x).hexdigest() for x in lines[:n]]==r['line_sha256']
    assert r['snapshot_script_sha256']==sha(out/'progress-snapshot.txt')
for stage in ['T04','T05']:
    for r in json.loads((e.BASE/'out'/stage/'artifact-archive-v1.json').read_text())['artifacts']:
        assert sha(e.BASE/r['archive'])==r['sha256']
        if stage=='T05':assert sha(e.BASE/r['original'])==r['sha256']
r=json.loads((out/'nested-error-v1.json').read_text())
assert r['all_equal'] and len(r['cases'])==20 and r['source_sha256']==sha(out/'nested-error-v1.txt')
for name in ['public-cpython-v2.log','public-pypy-v1.log']:
    text=(out/name).read_text();assert 'Ran 16 tests' in text and text.rstrip().endswith('OK')
assert 'running 11 tests' in (out/'cargo-v1.log').read_text()
for name in ['differential-v1','body-mutant-v1','return-mutant-v1']:
    r=json.loads((out/(name+'.json')).read_text())
    assert r['real_captures']==8 and sum(x['total'] for x in r['comparisons'].values())==2128
    assert sum(x['equal'] for x in r['comparisons'].values())==(2128 if name=='differential-v1' else 0)
for label in ['check','family-F40']:
    rows=[json.loads(p.read_text()) for p in (out/f'attest-{label}-v1').glob('*.json')]
    assert sum(r['native_calls'].get('0791:2FB1',0) for r in rows)>0
p=json.loads((out/'profile-after-v1.json').read_text())
assert len(p['routines'])==422 and p['routines']['0791:2FB1']['python_calls_or_resumes']==0
report=dict(pass_all=True,oracle_pristine=True,harness_sha256=harness,
            progress_prefixes_verified=[10,20,30],old_and_new_archives_verified=True,
            nested_error_cases=20,public_tests_each=16,rust_tests=11,
            registered_routine_addresses=422,native_routine_addresses=10,
            audit_source_sha256=sha(Path(__file__)))
e.write(out/'post-exit-audit-v1.json',report,before)
print('Additional oracle/harness, artifact, prefix and proof checks pass')
